## @exec 列存排序

`params(列存: Map, 目标列: Int)`

`列 = 列存["列"]`

`数据 = 列存["数据"]`

`行数 = 长度(数据[目标列])`

`顺序 = []`

`行号 = 0`

`while 行号 < 行数`

`顺序 = list.append(顺序, 行号)`

`行号 = 行号 + 1`

`end`

`下标 = 1`

`while 下标 < 行数`

`当前 = 下标`

`新顺序 = []`

`已插入 = false`

`位 = 0`

`while 位 < 长度(顺序)`

`已有 = 顺序[位]`

`if not 已插入 and 数据[目标列][已有] > 数据[目标列][当前]`

`新顺序 = list.append(新顺序, 当前)`

`新顺序 = list.append(新顺序, 已有)`

`已插入 = true`

`else`

`新顺序 = list.append(新顺序, 已有)`

`end`

`位 = 位 + 1`

`end`

`if not 已插入`

`新顺序 = list.append(新顺序, 当前)`

`else`

`忽略 = 0`

`end`

`顺序 = 新顺序`

`下标 = 下标 + 1`

`end`

`新数据 = []`

`列号 = 0`

`while 列号 < 长度(列)`

`新列 = []`

`位 = 0`

`while 位 < 行数`

`新列 = list.append(新列, 数据[列号][顺序[位]])`

`位 = 位 + 1`

`end`

`新数据 = list.append(新数据, 新列)`

`列号 = 列号 + 1`

`end`

`return {列: 列, 数据: 新数据}`

## @exec 表相等

`params(左: Table, 右: Table)`

`左列 = table.columns(左)`

`右列 = table.columns(右)`

`相等 = 长度(左列) == 长度(右列) and 长度(左) == 长度(右)`

`列号 = 0`

`while 列号 < 长度(左列) and 相等`

`if 左列[列号] != 右列[列号]`

`相等 = false`

`else`

`忽略 = 0`

`end`

`列号 = 列号 + 1`

`end`

`行号 = 0`

`while 行号 < 长度(左) and 相等`

`列号 = 0`

`while 列号 < 长度(左列) and 相等`

`if table.cell(左, 行号, 列号) != table.cell(右, 行号, 列号)`

`相等 = false`

`else`

`忽略 = 0`

`end`

`列号 = 列号 + 1`

`end`

`行号 = 行号 + 1`

`end`

`return 相等`
